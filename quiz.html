// assets/quiz.js let allTerms = []; let currentTerms = []; let currentQuestion = 0; let score = 0; let answer = ""; let totalQuestions = 10; let wrongQuestions = []; let retryMode = false; // =============================== // 기록 저장 키 // =============================== const QUIZ_RECORD_KEY = "term_quiz_record"; // =============================== // 퀴즈 기록 불러오기 // =============================== function getQuizRecord(){ const saved = localStorage.getItem(QUIZ_RECORD_KEY); if(!saved){ return { played:0, correct:0, bestScore:0, bestCombo:0 }; } try{ return JSON.parse(saved); } catch(e){ return { played:0, correct:0, bestScore:0, bestCombo:0 }; } } // =============================== // 기록 저장 // =============================== function saveQuizRecord(record){ localStorage.setItem( QUIZ_RECORD_KEY, JSON.stringify(record) ); } // =============================== // 현재 콤보 // =============================== let combo = 0; // =============================== // DOM // =============================== const quizType = document.getElementById("quiz-type"); const categorySelect = document.getElementById("category-select"); const question = document.getElementById("question"); const choices = document.getElementById("choices"); const result = document.getElementById("result"); const quizArea = document.getElementById("quiz-area"); const startArea = document.getElementById("start-area"); const nextBtn = document.getElementById("next-btn"); const quizCount = document.querySelector(".quiz-count"); // =============================== // 기록 표시 영역 생성 // =============================== function createRecordBox(){ const box = document.createElement("div"); box.id="quiz-record-box"; box.className="quiz-record-box"; startArea.before(box); updateRecordBox(); } function updateRecordBox(){ const box = document.getElementById( "quiz-record-box" ); if(!box)return; const record = getQuizRecord(); const accuracy = record.played > 0 ? Math.round( record.correct / record.played * 100 ) : 0; box.innerHTML = `

📊 나의 퀴즈 기록

총 풀이: ${record.played}문제

정답률: ${accuracy}%

최고 점수: ${record.bestScore}/${totalQuestions}

최고 콤보: ${record.bestCombo}

`; } // =============================== // 문제 불러오기 // =============================== async function loadTerms(){ const res = await fetch("terms.json"); allTerms = await res.json(); makeCategoryList(); createRecordBox(); } // =============================== // 카테고리 생성 // =============================== function makeCategoryList(){ const set = new Set(); allTerms.forEach(t=>{ (t.categories || []) .forEach(c=>set.add(c)); }); const labels = { stat:"통계", method:"연구방법론", tool:"측정·도구", ethics:"윤리·출판", physchem:"물리학·화학", bioearth:"생물학·지구과학", neuro:"뇌과학", medhealth:"의학·보건", psych:"심리학", socialecon:"사회과학·경제학", eng:"공학", cs:"컴퓨터과학·AI" }; [...set].forEach(cat=>{ const op = document.createElement("option"); op.value=cat; op.textContent = labels[cat] || cat; categorySelect.appendChild(op); }); } // =============================== // 퀴즈 시작 // =============================== document .getElementById("start-btn") .onclick=function(){ const cat = categorySelect.value; if(cat==="all"){ currentTerms = [...allTerms]; } else{ currentTerms = allTerms.filter(t=> (t.categories || []) .includes(cat) ); } shuffle(currentTerms); currentQuestion = 0; score = 0; combo = 0; totalQuestions = Math.min( 10, currentTerms.length ); wrongQuestions=[]; retryMode=false; startArea.hidden=true; quizArea.hidden=false; nextQuestion(); }; // =============================== // 문제 출제 // =============================== function nextQuestion(){ result.textContent=""; nextBtn.hidden=true; if( currentQuestion >= totalQuestions ){ finishQuiz(); return; } const term = currentTerms[currentQuestion]; let mode = quizType.value; if(mode==="random"){ mode = Math.random()>0.5 ? "definition" : "term"; } if(mode==="definition"){ answer = term.title_ko; question.textContent = term.definition; } else{ answer = term.definition; question.textContent = term.title_ko; } quizCount.textContent = `${currentQuestion+1} / ${totalQuestions}`; const options = [answer]; while(options.length < 4){ const randomTerm = allTerms[ Math.floor( Math.random() * allTerms.length ) ]; const option = mode==="definition" ? randomTerm.title_ko : randomTerm.definition; if( !options.includes(option) ){ options.push(option); } } shuffle(options); choices.innerHTML=""; options.forEach(op=>{ const div = document.createElement("div"); div.className="choice"; div.textContent=op; div.onclick=()=>{ check(div,op); }; choices.appendChild(div); }); updateProgress(); } // =============================== // 정답 확인 // =============================== function check(div,value){ [ ...choices.children ] .forEach(c=>{ c.onclick=null; }); const record = getQuizRecord(); record.played++; if(value===answer){ score++; combo++; record.correct++; if( combo > record.bestCombo ){ record.bestCombo = combo; } div.classList.add( "correct" ); result.textContent = `정답! 🔥 ${combo}연속 정답`; } else{ combo=0; wrongQuestions.push( currentTerms[currentQuestion] ); div.classList.add( "wrong" ); [ ...choices.children ] .forEach(c=>{ if( c.textContent===answer ){ c.classList.add( "correct" ); } }); result.textContent = `오답! 정답 : ${answer}`; } saveQuizRecord(record); currentQuestion++; updateProgress(); nextBtn.hidden=false; } // =============================== // 진행률 // =============================== function updateProgress(){ const bar = document.getElementById( "progress-bar" ); if(!bar)return; bar.style.width = ( currentQuestion / totalQuestions * 100 ) + "%"; } // =============================== // 종료 화면 // =============================== function finishQuiz(){ const record = getQuizRecord(); if( score > record.bestScore ){ record.bestScore = score; saveQuizRecord(record); } question.textContent = "퀴즈 종료!"; choices.innerHTML=""; const accuracy = Math.round( score / totalQuestions * 100 ); result.innerHTML = `

🎉 결과

점수: ${score}/${totalQuestions}

정답률: ${accuracy}%

최고 점수: ${record.bestScore}/${totalQuestions}

최고 콤보: ${record.bestCombo}

${ wrongQuestions.length ? ` ` : `

모든 문제 정답 🎉

` }
`; nextBtn.hidden=true; startArea.hidden=false; updateRecordBox(); const retryBtn = document.getElementById( "retry-btn" ); if(retryBtn){ retryBtn.onclick = startRetryQuiz; } } // =============================== // 오답 다시 풀기 // =============================== function startRetryQuiz(){ startArea.hidden=true; currentTerms = [...wrongQuestions]; wrongQuestions=[]; retryMode=true; currentQuestion=0; score=0; combo=0; totalQuestions = currentTerms.length; nextQuestion(); } // =============================== // 배열 섞기 // =============================== function shuffle(arr){ for( let i=arr.length-1; i>0; i-- ){ const j = Math.floor( Math.random() * (i+1) ); [ arr[i], arr[j] ] = [ arr[j], arr[i] ]; } } // 시작 loadTerms();
